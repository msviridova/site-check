-----------------------------------------------------------------
ТАБЛИЦА ДЛЯ ТЕКУЩИХ АКТИВНЫХ ПРОМПТОВ
-----------------------------------------------------------------
CREATE TABLE IF NOT EXISTS prompts (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  key_name      VARCHAR(64)     NOT NULL,         -- системный ключ: 'classify', 'creative_graphic', ...
  locale        VARCHAR(8)      NOT NULL DEFAULT 'ru',  -- 'ru'/'en' и т.п.
  version       INT             NOT NULL DEFAULT 1,     -- версионирование промптов
  description   VARCHAR(255)             DEFAULT NULL,  -- описание для людей
  text          LONGTEXT        NOT NULL,         -- текст промпта (с плейсхолдерами)
  is_active     TINYINT(1)      NOT NULL DEFAULT 1,
  updated_by    VARCHAR(64)              DEFAULT NULL,  -- кто правил
  created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  UNIQUE KEY uq_prompts_key_locale_active (key_name, locale, version)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
